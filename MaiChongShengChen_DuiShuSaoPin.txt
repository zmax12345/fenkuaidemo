clc
clear
close all
% 参数设置
Fks = 100000;             % 采样频率
T = 120;                 % 信号持续时间
t = 0:1/Fs:T-1/Fs;          % 时间向量
f_start = 5;           % 起始频率
f_end = 120;            % 结束频率
pulse_duration = 0.0005;  % 脉冲持续时间
R=log2(f_end/f_start) / T;%根据信号计算的扫频速度
% 生成瞬时频率的扫频信号
f_inst = f_start * 2.^(R * t);
figure,plot(t, f_inst);
xlabel('时间 (s)'); ylabel('频率 (Hz)');
title('指数扫频的频率变化曲线');

% 指数扫频信号公式
k = 2*pi*f_start / (R * log(2));
signal = sin(k * (2.^(R*t) - 1));
% 可视化
+-
+----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
figure;
subplot(2,1,1);
plot(t, signal);
title('指数扫频信号 s(t)');
xlabel('时间 (s)');
ylabel('幅值');
% 显示频率随时间变化
f_inst = f_start * 2.^(R * t);
subplot(2,1,2);
plot(t, f_inst);
title('扫频频率 f(t)');
xlabel('时间 (s)');
ylabel('频率 (Hz)');


%%
f_sample = 10*f_inst; % 瞬时频率

% 生成脉冲信号
pulse = zeros(size(t));
current_time = 0; % 当前时间
intervals = [];   % 存储脉冲间隔
pulse_count = 0;  % 脉冲计数器
pulse_times = []; % 存储脉冲中心时间
collected_signals = []; % 存储在脉冲时刻的扫频信号值

while current_time < T
    % 确保索引在有效范围内
    index = round(current_time * Fs);
    if index < 1
        index = 1; % 确保索引不小于1
    elseif index > length(f_sample)
        break; % 超出范围，退出循环
    end
    
    % 计算当前脉冲的间隔
    interval = 1 / f_sample(index);
    
    % 定义脉冲
    pulse_start = current_time;
    pulse_end = current_time + pulse_duration; % 脉冲持续时间
    
    if pulse_end < T
        pulse(t >= pulse_start & t <= pulse_end) = 10; % 定义脉冲
        pulse_times = [pulse_times; pulse_start + pulse_duration / 2]; % 存储脉冲中心时间
%         pulse_times = [pulse_times; pulse_start ]; % 存储脉冲中心时间
        % 收集在脉冲时刻的扫频信号值
        collected_signals = [collected_signals; signal(round(pulse_start * Fs)+1)]; 
        
        intervals = [intervals; interval]; % 记录脉冲间隔
        pulse_count = pulse_count + 1;     % 增加脉冲计数
    end
    
    % 更新当前时间
    current_time = pulse_end + interval; % 计算下一个脉冲的起始时间
end

figure
plot(t,pulse)
title('相机采集间隔');
xlabel('Time (s)');
ylabel('幅值/V');
xlim([0 T]); % 设置X轴范围
set(gca,'FontSize',13);

% 计算瞬时频率
instantaneous_frequencies = 1 ./ intervals; % 瞬时频率，单位为Hz

% 可视化瞬时频率
figure;
plot(pulse_times, instantaneous_frequencies);
hold on
plot(t,f_inst);
legend('采样信号瞬时频率','扫频信号瞬时频率')
title('扫频信号与采样信号瞬时频率对比');
xlabel('Time (s)');
ylabel('Frequency (Hz)');
xlim([0 T]); % 设置X轴范围
set(gca,'FontSize',13);


% 
figure;
plot(pulse_times, collected_signals, '.-');
hold on
plot(t, signal);
title('1kHz单频信号与非均匀采样信号');
xlabel('时间(s)');
ylabel('幅值')
legend('采样信号','原信号')
set(gca,'FontSize',13);


%%
t_new= 0:1/Fs:2*T-1/Fs;          % 时间向量
signal_new=zeros(size(t_new));
signal_new(length(signal)+1:end)=signal;
figure
plot(t_new,signal_new)
pulse_new=zeros(size(t_new));
pulse_new(length(pulse)+1:end)=pulse;
figure
plot(t_new,pulse_new)

%%
% 列名称
title={'xpos','value'};
%生成表格，按列生成
% VariableNames 参数用于设置列头
result_table=table(t_new,pulse_new,'VariableNames',title);
% 保存数据
writetable(result_table, 'D:\桌面文档\博士工作\对数扫频采样\Test_Waichufa.csv');
